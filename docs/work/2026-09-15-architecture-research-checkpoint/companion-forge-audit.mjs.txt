const client = new Client(
  { name: "forge-010-architecture-redteam", version: "1.0.0" },
  { versionNegotiation: { mode: "auto" } },
);
await client.connect(new StdioClientTransport({
  command: process.execPath,
  args: [`${companionRoot}/src/mcp-server.mjs`],
  cwd: companionRoot,
}));

async function call(name, args) {
  const response = await client.callTool({ name, arguments: args }, { timeout: 3_700_000, maxTotalTimeout: 3_700_000 });
  const body = JSON.parse(response.content?.[0]?.text || "{}");
  if (response.isError) throw Object.assign(new Error(body.message || `${name} failed`), { body });
  return body;
}
